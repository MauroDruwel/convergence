const CONVERGENCE_URL = window.location.origin;

export async function runAutonomousAgentTask(agentId: string, symbolId: string) {
  // 1. Acquire Edit-Time AST Symbol Lease (prevents token burn)
  const leaseRes = await fetch(`${CONVERGENCE_URL}/api/leases/request`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      agentId,
      symbolId,
      leaseType: "exclusive_write",
      ttlMs: 15000,
      intent: "Autonomous AST optimization & invariant hardening"
    })
  });

  const lease = await leaseRes.json();
  if (!lease.granted) {
    console.warn(`[HTTP 423] Symbol locked by ${lease.backpressure?.holder}. Waiting...`);
    return;
  }

  // 2. Push AST Symbol Mutation + Git Notes Provenance
  const commitRes = await fetch(`${CONVERGENCE_URL}/api/commit`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      agentId,
      filename: "index.ts",
      sourceCode: "// Updated AST node grafted by " + agentId,
      message: `feat(${symbolId}): autonomous CRDT symbol graft`
    })
  });

  return await commitRes.json();
}
